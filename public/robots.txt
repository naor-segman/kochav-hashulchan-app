# The signed-in product. Nothing here is reachable without an account, and a
# crawler that follows these links only ever reads a login screen.
Disallow: /app
Disallow: /account
Disallow: /admin
Disallow: /events/
Disallow: /login
Disallow: /signup
Disallow: /reset-password
Disallow: /auth/

# NOTE — the guest links (/invite/, /gift/, /rsvp/, /card/, /album/, /hostess/,
# /collab/, /save-the-date/, /invitation/) are deliberately NOT disallowed.
# Facebook's and WhatsApp's link-preview crawlers honour robots.txt, so blocking
# /invite/ would kill the per-event preview that netlify/edge-functions/
# invite-og.js exists to produce. They are kept out of search indexes instead
# by an `X-Robots-Tag: noindex` header on each of them (netlify.toml, audit
# 3.10), which is about search results, not previews.
